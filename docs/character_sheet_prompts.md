# Generate all 30 character sheet images
# Each prompt is passed to: npm run img -- "<prompt>"

for i in {1..6}; do
  for j in {1..5}; do
    echo "Generating Character $i - Cut $j..."

done

```bash

npm run img -- "[PROMPT FOR 1-2]"
# ... etc
```
